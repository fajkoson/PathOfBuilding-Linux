cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use ./build.sh or a Linux preset; build output belongs under out/build/")
endif()
include(cmake/BootstrapVcpkg.cmake)
project(PathOfBuildingLinux LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This build targets native Linux")
endif()
if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Use the linux-debug or linux-relwithdebinfo Clang preset")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(cmake/CompilerOptions.cmake)
include(cmake/Dependencies.cmake)
add_subdirectory(source/native/simplegraphic)
include(cmake/LinuxRuntime.cmake)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests/native)
endif()
